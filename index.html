import streamlit as st # Cấu hình trang web st.set_page_config(page_title="Hành Trình Khám Phá Thung Lũng Sức Khỏe", page_icon="🥗", layout="centered") # Khởi tạo trạng thái game if 'step' not in st.session_state: st.session_state.step = 0 if 'score' not in st.session_state: st.session_state.score = 0 if 'student_name' not in st.session_state: st.session_state.student_name = "" if 'student_class' not in st.session_state: st.session_state.student_class = "" # Tiêu đề game st.title("🥗 HÀNH TRÌNH KHÁM PHÁ THUNG LŨNG SỨC KHỎE") st.caption("Bài ôn tập Dinh dưỡng - Khối 4") # --- MÀN HÌNH BẮT ĐẦU --- if st.session_state.step == 0: st.markdown("### 🧙‍♂️ Thần Dinh Dưỡng chào em!") st.write("Chào mừng em đến với chuyến phiêu lưu dinh dưỡng. Hãy nhập thông tin để bắt đầu nhé!") name = st.text_input("Họ và tên của em:") cls = st.text_input("Lớp:") if st.button("🚀 Bắt đầu cuộc phiêu lưu!"): if name and cls: st.session_state.student_name = name st.session_state.student_class = cls st.session_state.step = 1 st.rerun() else: st.warning("Em vui lòng điền đầy đủ Tên và Lớp nhé!") # --- VÒNG 1: TRẮC NGHIỆM --- elif st.session_state.step == 1: st.header("🏆 Vòng 1: Đền Thờ Tri Thức") st.progress(0.25) st.markdown("**Câu 1:** Thực phẩm được chia thành mấy nhóm chất dinh dưỡng chính?") q1 = st.radio("Chọn câu trả lời:", [ "A. 2 nhóm (nhóm động vật và nhóm thực vật)", "B. 3 nhóm (nhóm đạm, nhóm béo, nhóm vitamin)", "C. 4 nhóm (nhóm bột đường, nhóm chất đạm, nhóm chất béo, nhóm vitamin và khoáng chất)" ], index=None) if st.button("Xác nhận Câu 1"): if q1 == "C. 4 nhóm (nhóm bột đường, nhóm chất đạm, nhóm chất béo, nhóm vitamin và khoáng chất)": st.success("Chính xác! Cơ thể chúng ta cần 4 nhóm chất chính. +10 điểm! 🎉") st.session_state.score += 10 st.session_state.step = 2 st.rerun() else: st.error("Chưa đúng rồi! Hãy nhớ lại bài học và thử chọn lại nhé!") elif st.session_state.step == 2: st.header("🏆 Vòng 1: Đền Thờ Tri Thức") st.progress(0.5) st.markdown("**Câu 2:** Nhóm thức ăn nào dưới đây có vai trò chính là xây dựng và phát triển cơ thể giúp em cao lớn hơn?") q2 = st.radio("Chọn câu trả lời:", [ "A. Nhóm chứa nhiều chất đạm.", "B. Nhóm chứa nhiều chất bột đường.", "C. Nhóm chứa nhiều vitamin và khoáng chất." ], index=None) if st.button("Xác nhận Câu 2"): if q2 == "A. Nhóm chứa nhiều chất đạm.": st.success("Tuyệt vời! Chất đạm giúp cơ thể cao lớn và khỏe mạnh. +10 điểm! 💪") st.session_state.score += 10 st.session_state.step = 3 st.rerun() else: st.error("Chưa chính xác! Nhóm nào giúp xây dựng cơ bắp và phát triển chiều cao nhỉ?") elif st.session_state.step == 3: st.header("🏆 Vòng 1: Đền Thờ Tri Thức") st.progress(0.75) st.markdown("**Câu 3:** Tại sao chúng ta cần phải ăn phối hợp nhiều loại thức ăn và thường xuyên thay đổi món?") q3 = st.radio("Chọn câu trả lời:", [ "A. Để thức ăn có màu sắc đẹp mắt hơn.", "B. Vì không có một loại thức ăn nào cung cấp đủ các chất dinh dưỡng cho nhu cầu của cơ thể.", "C. Để giúp chúng ta không bị khát nước." ], index=None) if st.button("Xác nhận Câu 3"): if q3 == "B. Vì không có một loại thức ăn nào cung cấp đủ các chất dinh dưỡng cho nhu cầu của cơ thể.": st.success("Đúng rồi! Ăn đa dạng giúp cơ thể nhận đủ dinh dưỡng. +10 điểm! 🌈") st.session_state.score += 10 st.session_state.step = 4 st.rerun() else: st.error("Thử lại xem nào! Hãy chọn lý do liên quan đến dinh dưỡng nhé!") elif st.session_state.step == 4: st.header("🏆 Vòng 1: Đền Thờ Tri Thức") st.progress(1.0) st.markdown("**Câu 4:** Trong các loại thực phẩm dưới đây, nhóm thực phẩm nào cung cấp nhiều vitamin và chất xơ nhất, giúp hệ tiêu hóa của em luôn khỏe mạnh?") q4 = st.radio("Chọn câu trả lời:", [ "A. Thịt heo, cá.", "B. Rau xà lách, xoài, măng cụt.", "C. Bánh mì, phô mai." ], index=None) if st.button("Xác nhận Câu 4"): if q4 == "B. Rau xà lách, xoài, măng cụt.": st.success("Xuất sắc! Rau củ quả chứa rất nhiều vitamin và chất xơ. +10 điểm! 🍎🥦") st.session_state.score += 10 st.session_state.step = 5 st.rerun() else: st.error("Chưa đúng! Tìm nhóm thực phẩm chứa rau và trái cây nhé!") # --- VÒNG 2: PHÂN LOẠI THỰC PHẨM --- elif st.session_state.step == 5: st.header("🧺 Vòng 2: Khu Rừng Phân Loại") st.write("Hãy giúp Thần Dinh Dưỡng xếp các thực phẩm sau vào đúng 4 nhóm nhé!") st.info("Danh sách: thịt heo, măng cụt, sữa, bánh mì, cá, xoài, phô mai, rau xà lách, đậu, trứng, tôm, thịt bò, quả bơ.") bot_duong = st.multiselect("1. Nhóm chứa nhiều chất BỘT ĐƯỜNG:", ["bánh mì", "thịt heo", "cá", "xoài", "sữa"]) dam = st.multiselect("2. Nhóm chứa nhiều CHẤT ĐẠM:", ["thịt heo", "cá", "tôm", "thịt bò", "đậu", "trứng", "sữa", "bánh mì"]) beo = st.multiselect("3. Nhóm chứa nhiều CHẤT BÉO:", ["phô mai", "quả bơ", "sữa", "măng cụt"]) vitamin = st.multiselect("4. Nhóm chứa nhiều VITAMIN & KHOÁNG CHẤT:", ["măng cụt", "xoài", "rau xà lách", "thịt bò"]) if st.button("Nộp bài Vòng 2"): # Kiểm tra điều kiện đơn giản if "bánh mì" in bot_duong and "thịt heo" in dam and "phô mai" in beo and "rau xà lách" in vitamin: st.success("Tối ưu lắm! Em đã phân loại thực phẩm rất chính xác. +30 điểm! 🧺✨") st.session_state.score += 30 st.session_state.step = 6 st.rerun() else: st.warning("Có vẻ một số thực phẩm chưa nằm đúng nhóm, em hãy kiểm tra kỹ lại nhé!") # --- VÒNG 3: VẬN DỤNG --- elif st.session_state.step == 6: st.header("🆘 Vòng 3: Giải Cứu Bạn Bi") st.write(""" **Tình huống:** Bạn Bi chỉ thích ăn thịt heo và ăn rất nhiều cơm, nhưng không bao giờ ăn xoài, măng cụt hay rau xà lách. """) ans1 = st.radio("1. Bữa ăn của bạn Bi đã cân bằng dinh dưỡng chưa?", ["Chưa cân bằng", "Đã cân bằng"], index=None) ans2 = st.text_area("2. Việc duy trì thói quen này sẽ ảnh hưởng như thế nào đến cơ thể bạn Bi? (Hãy viết lời khuyên cho Bi):") if st.button("Gửi lời khuyên"): if ans1 == "Chưa cân bằng" and len(ans2.strip()) > 10: st.success("Tuyệt vời! Lời khuyên của em rất có ý nghĩa. +30 điểm! 🌟") st.session_state.score += 30 st.session_state.step = 7 st.rerun() else: st.warning("Em hãy chọn câu trả lời và viết lời khuyên chi tiết hơn một chút nhé!") # --- MÀN HÌNH TỔNG KẾT --- elif st.session_state.step == 7: st.balloons() st.header("🎉 CHÚC MỪNG EM ĐÃ HOÀN THÀNH HÀNH TRÌNH!") st.write(f"**Họ và tên:** {st.session_state.student_name}") st.write(f"**Lớp:** {st.session_state.student_class}") st.subheader(f"🏆 Tổng điểm đạt được: {st.session_state.score} / 100 điểm") if st.session_state.score >= 80: st.markdown("### 🏅 Danh hiệu: **Nhà Thông Thái Dinh Dưỡng**") else: st.markdown("### 🏅 Danh hiệu: **Chủ Nhân Sức Khỏe Tương Lai**") if st.button("🔄 Chơi lại"): st.session_state.step = 0 st.session_state.score = 0 st.rerun()
